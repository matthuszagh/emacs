#+NAME: attr_wrap
#+header: :var data=""
#+header: :var orgwidth=""
#+header: :var htmlwidth=""
#+header: :var caption=""
#+header: :var name=""
#+header: :var latex=""
#+BEGIN_SRC sh :results output :exports none
  if [ -n "$caption" ]; then
    echo "#+CAPTION: $caption"
  fi
  if [ -n "$orgwidth" ]; then
    echo "#+ATTR_ORG: :width $orgwidth"
  fi
  if [ -n "$htmlwidth" ]; then
    echo "#+ATTR_HTML: :width $htmlwidth"
  fi
  if [ -n "$name" ]; then
    echo "#+NAME: $name"
  fi
  if [ -n "$latex" ]; then
    echo "#+ATTR_LATEX: $latex"
  fi
  # Remove all leading and trailing newlines. See
  # https://stackoverflow.com/a/23894449.
  echo "$data" | tac - | sed -e '/./,$!d' | tac | sed -e '/./,$!d'
  # echo "$data"
#+END_SRC

#+NAME: latexml_proc
#+header: :exports none
#+header: :var data=""
#+BEGIN_SRC sh :results output
  echo "#+begin_export html"
  latexmlc "literal:$data" --profile=fragment \
    --preload=siunitx.sty \
    --preload=mathtools.sty \
    --preload=math_op_local.tex\
    2>/dev/null
  echo "#+end_export"
#+END_SRC
